```typescript
export function parseAddress(fullAddress: string): {
  street: string;
  streetNumber: string;
  box: string | null;
  postalCode: string;
  city: string;
} {
  // Belgian address formats:
  // "Rue de la Loi 16, 1000 Bruxelles"
  // "Kerkstraat 42 bus 3, 9000 Gent"
  // "Avenue Louise 123/5, 1050 Ixelles"

  const trimmed = fullAddress.trim();

  // Split on the last comma to separate street from postal+city
  const lastCommaIdx = trimmed.lastIndexOf(",");
  let streetPart = trimmed;
  let postalCity = "";

  if (lastCommaIdx > -1) {
    streetPart = trimmed.substring(0, lastCommaIdx).trim();
    postalCity = trimmed.substring(lastCommaIdx + 1).trim();
  }

  // Parse postal code and city from "1000 Bruxelles"
  const postalMatch = postalCity.match(/^(\d{4})\s+(.+)$/);
  const postalCode = postalMatch ? postalMatch[1] : "";
  const city = postalMatch ? postalMatch[2].trim() : postalCity;

  // Parse street name, number, and optional box from "Rue de la Loi 16 bus 3" or "Rue de la Loi 16/3"
  let street = streetPart;
  let streetNumber = "";
  let box: string | null = null;

  // Try to find the street number (last number sequence in the street part, possibly followed by bus/boite)
  const numberMatch = streetPart.match(/^(.+?)\s+(\d+[\w]*)\s*(?:(?:bus|bte|boite|\/)\s*(\w+))?$/i);
  if (numberMatch) {
    street = numberMatch[1].trim();
    streetNumber = numberMatch[2];
    box = numberMatch[3] || null;
  }

  return { street, streetNumber, box, postalCode, city };
}
```

**Step 4: Region guesser from postal code:**

```typescript
export function guessRegion(postalCode: string): "flanders" | "wallonia" | "brussels" {
  const code = parseInt(postalCode, 10);
  if (code >= 1000 && code <= 1299) return "brussels";
  // Flemish postal codes: 1500-3999, 8000-9999
  if ((code >= 1500 && code <= 3999) || (code >= 8000 && code <= 9999)) return "flanders";
  // Walloon: 1300-1499, 4000-7999
  return "wallonia";
}
```

```typescript
export function mapLeaseType(smovinType?: string): "residential_short" | "residential_long" | "commercial" | "student" | "residential_lifetime" {
```

**Step 6: Date parser (handles DD/MM/YYYY and ISO):**

```typescript
export function parseDate(dateStr: string): string {
  // Handle DD/MM/YYYY format common in Belgian apps
  const ddmmyyyy = dateStr.match(/^(\d{1,2})[\/\-.](\d{1,2})[\/\-.](\d{4})$/);
  if (ddmmyyyy) {
    const day = ddmmyyyy[1].padStart(2, "0");
    const month = ddmmyyyy[2].padStart(2, "0");
    return `${ddmmyyyy[3]}-${month}-${day}`;
  }
  // Already ISO format or close to it
  return dateStr.substring(0, 10);
}
```

**Step 7: Amount parser (handles European comma decimals):**

```typescript
export function parseAmount(amountStr: string): string {
  // "1.250,00" -> "1250.00" or "850.00" -> "850.00"
  const cleaned = amountStr.replace(/[^\d,.\-]/g, "");
  // If contains both comma and dot, comma is decimal separator if it comes last
  if (cleaned.includes(",") && cleaned.includes(".")) {
    if (cleaned.lastIndexOf(",") > cleaned.lastIndexOf(".")) {
      // European: 1.250,00
      return cleaned.replace(/\./g, "").replace(",", ".");
    }
    // US: 1,250.00
    return cleaned.replace(/,/g, "");
  }
  // Only comma: treat as decimal separator
  if (cleaned.includes(",")) {
    return cleaned.replace(",", ".");
  }
  return cleaned;
}
```

```typescript

  maxRetriesPerRequest: null,

  defaultJobOptions: {
    attempts: 2,
    backoff: { type: "exponential", delay: 10000 },

    removeOnFail: { count: 50 },
```

**Worker processor logic:**

1. Load session from DB by sessionId
2. Parse discoveredData JSON and selectedProperties JSON (array of indices)
3. Filter discoveredData to only selected properties
4. For each selected property:
   a. **Duplicate check for property (D-06):** Query `properties` table WHERE `ownerId = session.userId` AND address columns match (street + streetNumber + postalCode + city). Use `parseAddress()` on the Smovin address to get components. If a match exists, skip this property and increment skipped counter.
   b. **Insert property:** Use `mapSmovinProperty()` to create insert values, then `db.insert(properties).values(mappedProp)`
   c. **For each tenant in the property:**
      - **Duplicate check for tenant (D-06):** Query `tenants` table WHERE `ownerId = session.userId` AND `email = tenant.email` (only if tenant has email). If match exists, use the existing tenant ID instead of creating new one.
      - If no duplicate: `mapSmovinTenant()` then `db.insert(tenants).values(mappedTenant)`
   d. **For each lease in the property:**
      - `mapSmovinLease()` then `db.insert(leases).values(mappedLease)`
      - Link tenants to lease via `db.insert(leaseTenants).values({ leaseId: mappedLease.id, tenantId: tenantId, isPrimary: index === 0 })`
   e. **For each payment in the property:**
      - Need a leaseId to link to. Use the first lease's ID for the property (or skip payments if no lease).
      - `mapSmovinPayment()` then `db.insert(payments).values(mappedPayment)`
   f. **Update progress:** After each property, update session:

```typescript
    .where(and(
```
