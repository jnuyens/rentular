<!-- API endpoints (from Plan 01/02) that the frontend calls -->

GET /api/v1/communications?propertyId=&tenantId=&type=&page=1&perPage=50
Response: { data: Communication[], meta: { total, page, perPage } }

GET /api/v1/communications/:id
Response: { data: Communication }

POST /api/v1/communications/:id/resend
Response: { message: "Communication re-queued for delivery" }

POST /api/v1/communications/send
Body: { leaseId, channel, subject?, body }
Response: { message: "Message queued for delivery" }

GET /api/v1/settings/smtp
Response: { data: { host, port, username, fromAddress, fromName, verified, lastVerifiedAt, hasPassword } | null }

PUT /api/v1/settings/smtp
Body: { host, port, username, password, fromAddress, fromName? }

POST /api/v1/settings/smtp/test
Body: { host, port, username, password, fromAddress }
Response: { success: boolean, message?: string, error?: string }

DELETE /api/v1/settings/smtp
Response: { message: "SMTP settings removed" }

GET /api/v1/properties (for filter dropdown)
Response: { data: Property[] }

GET /api/v1/tenants (for filter dropdown)
Response: { data: Tenant[] }

    Build per the UI-SPEC. This is a `"use client"` component using `useTranslations("communications")` from next-intl.

    Key implementation details:

    a) **State:**
    - `communications`: array of communication objects
    - `loading`: boolean
    - `error`: string | null
    - `expandedId`: string | null (only one row expanded at a time -- accordion per UI-SPEC)
    - `propertyFilter`: string (property ID or empty for all)
    - `tenantFilter`: string (tenant ID or empty for all)
    - `typeFilter`: string (type enum or empty for all)
    - `properties`: array for property dropdown
    - `tenants`: array for tenant dropdown

    b) **Data fetching:**
    - On mount and when filters change, fetch `GET /api/v1/communications?propertyId=${propertyFilter}&tenantId=${tenantFilter}&type=${typeFilter}&perPage=50`
    - On mount, fetch properties and tenants for filter dropdowns: `GET /api/v1/properties` and `GET /api/v1/tenants`

    ```tsx
    <h1 className="text-2xl font-semibold">{t("title")}</h1>
    ```

    - Three native `<select>` elements in a `div.flex.gap-2.items-center.mb-6`

    - Tenant filter: options from tenants list, default "All tenants"

    ```typescript
      if (!smtpSettings.host) return; // All-or-nothing: if host is empty, treat as "no custom SMTP"

      try {
        const body: Record<string, unknown> = {
          host: smtpSettings.host,
          port: smtpSettings.port,
          username: smtpSettings.username,
          fromAddress: smtpSettings.fromAddress,
          fromName: smtpSettings.fromName || undefined,
        };
    ```
