    **1. Key comparison audit:**

    Write and run a Node.js script to compare all 4 locale files and identify missing keys:
    ```bash
    node -e "
      const en = require('./apps/web/messages/en/common.json');
      const nl = require('./apps/web/messages/nl/common.json');
      const fr = require('./apps/web/messages/fr/common.json');
      const de = require('./apps/web/messages/de/common.json');

      const allKeys = new Set([...Object.keys(en), ...Object.keys(nl), ...Object.keys(fr), ...Object.keys(de)]);
      const locales = { en, nl, fr, de };
      let missing = 0;

      for (const key of [...allKeys].sort()) {
        const absent = Object.entries(locales)
          .filter(([, obj]) => !(key in obj))
          .map(([name]) => name);
        if (absent.length > 0) {
          console.log('MISSING: ' + key + ' in ' + absent.join(', '));
          missing++;
        }
      }

      console.log('---');
      console.log('Total keys: ' + allKeys.size);
      console.log('EN: ' + Object.keys(en).length);
      console.log('NL: ' + Object.keys(nl).length);
      console.log('FR: ' + Object.keys(fr).length);
      console.log('DE: ' + Object.keys(de).length);
      console.log('Missing: ' + missing);

      if (missing > 0) process.exit(1);
    "
    ```
